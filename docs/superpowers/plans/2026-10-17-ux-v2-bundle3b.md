> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 결과 섹션에 이미지 블록(Mock SVG 자리표시)을 넣고, 과제 폴더의 샘플 근거 자료를 검토해 결과에 점검 기록과 출처를 붙인 새 버전(`evidence`)을 만든다.

**Architecture:** 백엔드는 섹션 블록을 `TextBlock | ImageBlock`(type 구분)으로 넓히고, 파일은 `AssetStore`(메모리·로컬 폴더) 뒤에 둔다. 이미지는 별도 제공자 `ImageProvider`의 `image` 단계가 정책 `image_generation`을 켠 과제에서만 만든다(실패해도 결과는 살리고 경고 하나). 근거 자료는 `evidence.yaml`을 읽는 `find_evidence` 단계가 내고, 반영은 문구 수정처럼 직전 버전의 기준을 이어받는 파생 버전이다. 프론트는 이미지를 `X-Client-Id`를 붙여 받아 data URL로 들고(브라우저 `<img>`는 헤더를 못 보낸다), 문서·내려받기·미리보기가 같은 값을 쓴다.

**Tech Stack:** FastAPI·Pydantic v2·Jinja2·PyYAML(백엔드), React 19·Vite·TypeScript·openapi-fetch·vitest(프론트).

## 결정 사항 (컨트롤러)

1. **근거 기록의 자리**: 자료의 `applies_to`는 섹션 종류(`diagnosis`·`suggestion`·`action`)다. 반영할 때 그 종류의 첫 섹션 order를 `section_order`로 정하고(없으면 `null` = 문서 끝), 화면은 그 섹션 바로 뒤에 기록 상자를 그린다. 섹션 order는 LLM이 정하므로 종류로 고른다.
2. **근거 기록은 결과에 산다**: `GenerationResult.evidence: list[EvidenceRecord]`. 문구 수정·근거 반영 버전은 결과를 복사하므로 그대로 남고, 다시 만든 초안(`regenerate`·`supplement`)은 코드가 이전 결과의 기록을 새 섹션에 다시 자리 잡아 옮긴다(`carry_evidence`). 세션 필드는 늘리지 않는다. 생성 프롬프트에는 싣지 않는다(점검 문구라 글에 섞을 내용이 아니다).
3. **같은 자료를 다시 반영**하면 버전을 만들지 않는다(문구 수정의 "같은 글"과 같다). 점검 문구(`record`)가 없는 자료(배경 자료)는 반영할 수 없다(422).
4. **반영 여부는 프론트가 결과에서 계산**한다(`result.evidence`의 `source_id`). 자료 목록 응답에는 반영 여부를 싣지 않아 목록이 낡을 일이 없다.
5. **근거 버튼 표시**: `TaskSummary.evidence_sources`(그 과제의 샘플 자료 수). 0이면 결과 화면에 '근거 자료 검토' 버튼이 없다. 지금은 상품기획만 자료가 있다(디자인과 같다).
6. **이미지 한 장**: `image_generation`을 켠 과제에서 생성마다 대표 섹션(첫 `suggestion`, 없으면 첫 섹션) 끝에 한 장. 이미지 실패는 생성 실패가 아니다(경고 `image_unavailable`). 생성 뒤 다른 단계가 실패하면 저장한 파일이 남을 수 있다(정리는 배포 때 다시 본다).
7. **문구 수정은 이미지 블록을 지키고** 글 블록만 바꾼다(첫 글 블록 자리에 새 글 하나).
8. **이미지 전달**: `GET /api/v1/coaching/assets/{asset_id}`가 바이트를 낸다(owner 검사, 남의 것·없는 것은 404). 프론트는 헤더를 붙여 받아 data URL로 바꾼다. SVG는 `<img>`로만 그리므로 스크립트가 돌지 않고, 응답에는 `X-Content-Type-Options: nosniff`와 `Content-Security-Policy: default-src 'none'; style-src 'unsafe-inline'; sandbox`를 붙인다.
9. **이미지 불러오기는 화면 불러오기와 따로**다: 요청 번호(`inflight`)를 쓰지 않아 버튼을 막지 않고, 실패해도 알림 없이 대체 글(alt)로 둔다.
10. 원문 링크는 `http(s)://`만 그린다(백엔드 YAML 검증 + 프론트 `safeUrl`). 나중에 웹 검색 결과가 와도 같은 자리에서 걸러진다.

```python
def _replace_text(blocks: Sequence[TextBlock | ImageBlock], text: str) -> list[TextBlock | ImageBlock]:
    """[순수] 글 블록들을 새 글 하나로 바꾼다. 첫 글 블록 자리에 두고, 이미지 블록은 순서대로 지킨다(글이 없었으면 맨 앞)."""
    replaced: list[TextBlock | ImageBlock] = []
    placed = False
    for block in blocks:
        if block.type != "text":
            replaced.append(block.model_copy())
        elif not placed:
            replaced.append(TextBlock(text=text))
            placed = True
    return replaced if placed else [TextBlock(text=text), *replaced]
```

```python
"""결과 이미지 같은 파일의 저장소. 메모리(테스트)와 로컬 폴더(서버를 다시 켜도 남는다) 두 가지다.

세션 JSON에는 파일 id만 두고 바이트는 여기에 둔다. 주인(owner_id)을 함께 적어 내려줄 때 검사한다.
"""
import json
import re
from dataclasses import dataclass
from pathlib import Path
from typing import Protocol
from uuid import uuid4

from app.core.store import StoreKind

IMAGE_MIMES = frozenset({"image/svg+xml", "image/png", "image/jpeg", "image/webp"})
_ASSET_ID = re.compile(r"asset_[0-9a-f]{32}")


class UnsupportedAssetType(ValueError):
    """이미지가 아닌 형식이다(제공자 구현 오류)."""


@dataclass(frozen=True)
class Asset:
    owner_id: str
    mime: str
    data: bytes


class AssetStore(Protocol):
    """파일 저장소 계약. put은 새 id를 돌려주고(이미지 형식만), get은 없거나 형식이 틀린 id면 None이다."""

    def put(self, owner_id: str, data: bytes, mime: str) -> str: ...

    def get(self, asset_id: str) -> Asset | None: ...


def _new_id(mime: str) -> str:
    if mime not in IMAGE_MIMES:
        raise UnsupportedAssetType(f"이미지가 아닌 형식입니다: {mime}")
    return f"asset_{uuid4().hex}"


class InMemoryAssetStore:
    """서버를 재시작하면 사라진다. 테스트의 기본값이다."""

    def __init__(self) -> None:
        self._assets: dict[str, Asset] = {}

    def put(self, owner_id: str, data: bytes, mime: str) -> str:
        asset_id = _new_id(mime)
        self._assets[asset_id] = Asset(owner_id=owner_id, mime=mime, data=data)
        return asset_id

    def get(self, asset_id: str) -> Asset | None:
        return self._assets.get(asset_id)


class LocalAssetStore:
    """폴더 하나에 파일마다 바이트(`{id}.bin`)와 정보(`{id}.json`: 주인·형식)를 둔다. id 형식만 받아 폴더 밖을 읽지 않는다."""

    def __init__(self, root: Path) -> None:
        root.mkdir(parents=True, exist_ok=True)
        self._root = root

    def put(self, owner_id: str, data: bytes, mime: str) -> str:
        asset_id = _new_id(mime)
        (self._root / f"{asset_id}.bin").write_bytes(data)
        (self._root / f"{asset_id}.json").write_text(json.dumps({"owner_id": owner_id, "mime": mime}), encoding="utf-8")
        return asset_id

    def get(self, asset_id: str) -> Asset | None:
        if not _ASSET_ID.fullmatch(asset_id):
            return None
        meta = self._root / f"{asset_id}.json"
        if not meta.exists():
            return None
        info = json.loads(meta.read_text(encoding="utf-8"))
        return Asset(owner_id=info["owner_id"], mime=info["mime"], data=(self._root / f"{asset_id}.bin").read_bytes())


def build_asset_store(kind: StoreKind, root: Path) -> AssetStore:
    """세션 저장소(STORE)와 짝을 맞춘다: sqlite면 로컬 폴더(ASSET_DIR), memory면 메모리."""
    return LocalAssetStore(root) if kind == "sqlite" else InMemoryAssetStore()
```

```python
                output={"asset_id": asset_id, "mime": image.mime, "bytes": len(image.data)},
                provider=self.images.name,
                model=image.model,
            )
        )
        block = ImageBlock(
            asset_id=asset_id, alt=f"{target.heading} 참고 이미지", caption=IMAGE_CAPTION, origin=ImageOrigin.AI_GENERATED
        )
        return with_image(result, target.order, block), []
```

```python
class EvidenceRecord(ContractModel):
    """결과에 붙인 근거 검토 기록(점검 문구와 출처). 그 종류의 첫 섹션 뒤에 보이고, 자리가 없으면 문서 끝이다."""

    source_id: str
    title: str  # 점검 항목 이름
    text: str
    note: str  # 출처 옆 한 줄(예: 제품 시험 전)
    source_title: str
    publisher: str
    url: str
    section_kind: CardType
    section_order: int | None


def section_order_for(sections: Sequence[ResultSection], kind: CardType) -> int | None:
    """[순수] 그 종류의 첫 섹션 order. 없으면 None."""
    return next((s.order for s in sections if s.kind is kind), None)


def carry_evidence(records: Sequence[EvidenceRecord], sections: Sequence[ResultSection]) -> list[EvidenceRecord]:
    """[순수] 다시 만든 결과로 옮기는 근거 기록. 섹션이 바뀌었으니 종류로 자리를 다시 잡는다."""
    return [r.model_copy(update={"section_order": section_order_for(sections, r.section_kind)}) for r in records]
```

```python
"""근거 자료. 지금은 과제 폴더 evidence.yaml(샘플)을 읽는다. 실제 웹 검색을 붙이면 find_evidence만 바꾼다.

모르는 키(오타)는 읽을 때 거부한다. 원문 주소는 http(s)만 받는다(문서·화면 링크로 그대로 쓰이므로).
"""
from collections.abc import Sequence
from pathlib import Path

import yaml
from pydantic import BaseModel, ConfigDict, Field, model_validator

from app.core.results import CardType, EvidenceRecord, ResultSection, section_order_for


class EvidenceProposal(BaseModel):
    """결과에 붙일 점검 문구. 배경 자료에는 없다."""

    model_config = ConfigDict(extra="forbid")

    title: str
    text: str
    note: str = ""  # 출처 옆 한 줄(예: 제품 시험 전)
    caution: str = ""  # 반영하기 전에 보여 줄 안내


class EvidenceSource(BaseModel):
    model_config = ConfigDict(extra="forbid")

    id: str = Field(pattern=r"^[a-z0-9_]+$")
    title: str
    publisher: str
    meta: str
    summary: str
    scope: str  # 적용 범위(이 결과의 검증이 아님을 밝힌다)
    url: str = Field(pattern=r"^https?://")
    applies_to: CardType  # 기록을 둘 섹션 종류
    record: EvidenceProposal | None = None


class _Catalog(BaseModel):
    model_config = ConfigDict(extra="forbid")

    sources: list[EvidenceSource]

    @model_validator(mode="after")
    def _unique_ids(self) -> "_Catalog":
        ids = [s.id for s in self.sources]
        repeated = sorted({i for i in ids if ids.count(i) > 1})
        if repeated:
            raise ValueError(f"근거 자료 id가 중복됩니다: {', '.join(repeated)}")
        return self


def load_evidence(path: Path) -> list[EvidenceSource]:
    """과제의 샘플 근거 자료. 파일이 없으면 자료가 없는 과제다."""
    if not path.exists():
        return []
    with path.open(encoding="utf-8") as fp:
        return _Catalog.model_validate(yaml.safe_load(fp)).sources


def evidence_record(source: EvidenceSource, sections: Sequence[ResultSection]) -> EvidenceRecord:
    """[순수] 자료의 점검 문구를 결과에 붙일 기록으로. 점검 문구가 없는 자료는 부르는 쪽이 거른다."""
    if source.record is None:
        raise ValueError(f"점검 문구가 없는 자료입니다: {source.id}")
    return EvidenceRecord(
        source_id=source.id,
        title=source.record.title,
        text=source.record.text,
        note=source.record.note,
        source_title=source.title,
        publisher=source.publisher,
        url=source.url,
        section_kind=source.applies_to,
        section_order=section_order_for(sections, source.applies_to),
    )
```

```python
    def _add_derived_version(self, state: SessionState, reason: VersionReason, result: GenerationResult) -> None:
        """[코드] 직전 버전의 기준(사실·가정·미정·보완 정보)을 이어받고 결과만 바꾼 새 버전(문구 수정·근거 반영).

        LLM을 부르지 않는다. 코드 검증은 지금 세션의 바뀐 답이 아니라 그 기준으로 다시 돈다. 상태(FAILED 포함)와
        result_stale은 건드리지 않는다.
        """
        latest = state.versions[-1]
        basis = state.model_copy(
            update={
                "facts": latest.facts,
                "assumptions": latest.assumptions,
                "unknowns": latest.unknowns,
                "supplements": latest.supplements,
            }
        )
        report = ValidationReport(issues=collect_issues([*COMMON_VALIDATORS, *self.validators], basis, result, self.policy))
        state.versions.append(latest.derived(len(state.versions) + 1, reason, result, report))
        state.result = result
        state.validation = report
```

```python
    def find_evidence(self, state: SessionState) -> list[EvidenceSource]:
        """[자료] 결과를 검토할 근거 자료. 지금은 과제 폴더 evidence.yaml(샘플)이고, 웹 검색을 붙이면 이 단계만 바꾼다."""
        return list(self.evidence)

    def apply_evidence(self, state: SessionState, source_id: str) -> bool:
        """[코드] 자료의 점검 문구와 출처를 결과에 붙인 새 버전(evidence)을 만든다. LLM을 부르지 않는다.

        직전 버전의 기준을 이어받는다(_add_derived_version). 이미 반영한 자료면 버전 없이 True. 결과가 없거나 그런
        자료가 없으면 False(세션은 그대로). 점검 문구가 없는 자료(배경 자료)면 HarnessError.
        """
        if state.result is None or not state.versions:
            return False
        source = next((s for s in self.find_evidence(state) if s.id == source_id), None)
        if source is None:
            return False
        if source.record is None:
            raise HarnessError(EVIDENCE_NO_RECORD_MESSAGE)
        if any(r.source_id == source_id for r in state.result.evidence):
            return True
        record = evidence_record(source, state.result.sections)
        self._add_derived_version(state, VersionReason.EVIDENCE, with_evidence(state.result, record))
        return True
```

```python
    """결과를 검토할 근거 자료(지금은 샘플). 결과가 있어야 검토할 수 있다."""
    state, harness = _load(registry, store, session_id, owner)

    responses=error_responses(404, 422),

def apply_evidence(
    session_id: str, source_id: str, registry: Registry, store: Store, locks: Locks, owner: Owner
```

```ts
const CHUNK = 0x8000;

/** [순수] 바이트를 base64 data URL로. 조각으로 나눠 String.fromCharCode의 인자 수 한계를 넘지 않는다. */
export function bytesToDataUrl(mime: string, bytes: Uint8Array): string {
  let binary = "";
  for (let i = 0; i < bytes.length; i += CHUNK) {
    binary += String.fromCharCode(...bytes.subarray(i, i + CHUNK));
  }
  return `data:${mime};base64,${btoa(binary)}`;
}

/** 받은 이미지(Blob)를 data URL로. 화면 <img>와 내려받는 문서가 같은 값을 쓴다. */
export async function blobToDataUrl(blob: Blob): Promise<string> {
  return bytesToDataUrl(blob.type, new Uint8Array(await blob.arrayBuffer()));
}
```

```ts
/** asset_id → data URL. null은 불러오는 중이거나 불러오지 못했다(대체 글을 보인다). */
export type Images = Record<string, string | null>;

export type DocumentBlock =
  | { kind: "text"; key: string; text: string }
  | { kind: "image"; key: string; src: string | null; alt: string; caption: string };

export type DocumentRecord = { key: string; label: string; text: string; source: string; note: string; url: string | null };

export type DocumentSection = { key: string; number: string; heading: string; tag: string; blocks: DocumentBlock[]; records: DocumentRecord[] };

function documentBlocks(section: ResultSection, images: Images): DocumentBlock[] {
  return section.blocks.map((block, index) =>
    block.type === "text"
      ? { kind: "text", key: String(index), text: block.text }
      : { kind: "image", key: String(index), src: images[block.asset_id] ?? null, alt: block.alt, caption: block.caption },
  );
}

function documentRecord(record: EvidenceRecord): DocumentRecord {
  return {
    key: record.source_id,
    label: `근거 검토 반영 · ${record.title}`,
    text: record.text,
    source: `출처: ${record.publisher} · ${record.source_title}`,
    note: record.note,
    url: safeUrl(record.url),
  };
}

/** [순수] 결과 섹션을 문서 섹션(순서대로 01, 02 …)으로. 근거 기록은 자리 잡은 섹션에 붙인다. */
export function documentSections(sections: ResultSection[], evidence: EvidenceRecord[], images: Images): DocumentSection[] {
  return [...sections]
    .sort((a, b) => a.order - b.order)
    .map((section, index) => ({
      key: `${section.kind}-${section.order}`,
      number: String(index + 1).padStart(2, "0"),
      heading: section.heading,
      tag: cardLabel(section.kind),
      blocks: documentBlocks(section, images),
      records: evidence.filter((r) => r.section_order === section.order).map(documentRecord),
    }));
}

/** [순수] 자리 잡을 섹션이 없는 근거 기록(문서 끝에 둔다). */
export function trailingRecords(sections: ResultSection[], evidence: EvidenceRecord[]): DocumentRecord[] {
  const orders = new Set(sections.map((s) => s.order));
  return evidence.filter((r) => r.section_order === null || !orders.has(r.section_order)).map(documentRecord);
}
```

```ts
function blockHtml(block: DocumentBlock): string {
  const e = escapeHtml;
  if (block.kind === "text") return `<p>${e(block.text)}</p>`;
  if (block.src !== null && block.src.startsWith("data:image/")) {
    const caption = block.caption ? `<figcaption>${e(block.caption)}</figcaption>` : "";
    return `<figure><img src="${e(block.src)}" alt="${e(block.alt)}">${caption}</figure>`;
  }
  return `<p class="note">[이미지: ${e(block.alt)}]</p>`;
}

function recordHtml(record: DocumentRecord): string {
  const e = escapeHtml;
  const source = record.url ? `<a href="${e(record.url)}">${e(record.source)}</a>` : e(record.source);
  const note = record.note ? ` · ${e(record.note)}` : "";
  return `<div class="record"><p><strong>${e(record.label)}</strong></p><p>${e(record.text)}</p><p class="note">${source}${note}</p></div>`;
}
```
