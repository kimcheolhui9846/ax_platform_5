# 디자인 V2 묶음 3a (결과 섹션·문구 수정·미확인 정보 보완) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 결과를 섹션·블록 구조로 바꾸고, 결과 문구를 직접 고친 새 버전과 미확인 정보(보완 정보)를 입력해 초안에 반영한 새 버전을 만든다.

**Architecture:** LLM이 채우는 출력(`TaskResult.cards`)은 그대로 두고, 저장하는 결과(`GenerationResult`)만 섹션(`ResultSection`)과 블록(`TextBlock`)으로 바꾼다(묶음 3b의 `ImageBlock`이 같은 자리에 들어간다). 문구 수정은 LLM 없이 섹션 글을 바꾼 버전(`edit`)을 덧붙이고, 보완 정보는 세션의 `supplements`에 사용자 입력으로 두었다가 다음 생성에 반영한다(`supplement`). 프론트는 결과 오른쪽에 보완 패널(대화상자)과 문구 수정 화면을 더한다.

**Tech Stack:** FastAPI · Pydantic v2 · Jinja2 · pytest, React 19 · Vite 8 · TypeScript 5.9 · vitest 5

## 정한 것

1. **LLM 출력은 카드 그대로.** `to_generation_result`가 카드를 섹션(글 블록 하나)으로 바꾼다. 프롬프트·Mock·과제 스키마는 바뀌지 않는다. 이미 저장된 결과(`cards`)는 읽을 때 섹션으로 바꾼다.
2. **문구 수정은 섹션 하나의 글을 바꾸고 완성본(`full_text`)을 다시 만든다.** 새 버전(`edit`)은 직전 버전의 기준(사실·가정·미정·보완)을 그대로 이어받는다. 코드 검증만 다시 돈다(LLM 검토는 부르지 않는다). 결과의 최신 여부(`result_stale`)는 건드리지 않는다.
3. **보완 정보는 정책의 `supplement_fields`(필드와 이름이 겹치지 않는 별도 항목)다.** 세션의 `supplements`에 출처 `user_supplement`("사용자 입력 · 검증 전")로 두고, 질문 흐름·준비도에는 넣지 않는다. 생성 프롬프트의 세션 맥락에 "보완 정보"로 실린다.
4. **생성 이유는 코드가 정한다.** 답(사실·가정·미정)은 그대로이고 보완 정보만 바뀌었으면 `supplement`, 아니면 `regenerate`(첫 생성은 `first`). API는 바뀌지 않는다.
5. **문구 수정 화면은 섹션 번호로 연다**(`#/s/{id}/edit/{order}`). 결과 화면의 '소개 문구 수정하기'는 첫 섹션을 연다.

```python
class TextBlock(ContractModel):
    """결과 섹션의 글 블록. 이미지 블록(묶음 3b)과 type으로 구분한다."""

    type: Literal["text"] = "text"
    text: str


class ResultSection(ContractModel):
    """결과 문서의 섹션 하나(진단·제안·실행 과제). 내용은 블록 목록이다."""

    order: int
    heading: str
    kind: CardType
    blocks: list[TextBlock]


def sections_from_cards(cards: Sequence[ResultCard]) -> list[ResultSection]:
    """[순수] LLM이 쓴 카드를 순서대로 섹션으로 바꾼다(본문은 글 블록 하나)."""
    return [
        ResultSection(order=c.order, heading=c.title, kind=c.card_type, blocks=[TextBlock(text=c.body)])
        for c in sorted(cards, key=lambda c: c.order)
    ]


def section_text(section: ResultSection) -> str:
    """[순수] 섹션의 글 블록을 이어 붙인 본문."""
    return "\n\n".join(block.text for block in section.blocks if block.type == "text")


def compose_full_text(sections: Sequence[ResultSection]) -> str:
    """[순수] 섹션들로 다시 만든 완성본(문구를 고친 뒤 검증·복사에 쓴다)."""
    return "\n\n".join(f"{s.heading}\n{section_text(s)}" for s in sections)
```

```python
class GenerationResult(ContractModel):
    """세션에 저장되는 결과. content는 과제별 스키마를 통과한 값의 dict다."""

    sections: list[ResultSection]
    assumptions: list[ResultAssumption] = Field(default_factory=list)
    full_text: str
    content: dict[str, Any]
    generated_at: datetime = Field(default_factory=utcnow)

    @model_validator(mode="before")
    @classmethod
    def _sections_from_legacy_cards(cls, data: Any) -> Any:
        """섹션이 생기기 전에 저장한 결과(cards)는 섹션으로 읽는다."""
        if isinstance(data, dict) and "sections" not in data and "cards" in data:
            cards = [ResultCard.model_validate(c) for c in data["cards"]]
            data = {k: v for k, v in data.items() if k != "cards"} | {"sections": sections_from_cards(cards)}
        return data
```

```ts
/** [순수] 섹션의 글 블록을 이어 붙인 본문. */
export function sectionText(section: Pick<ResultSection, "blocks">): string {
  return section.blocks
    .filter((block) => block.type === "text")
    .map((block) => block.text)
    .join("\n\n");
}

/** [순수] 결과 섹션을 문서 섹션(순서대로 01, 02 …)으로. */
export function documentSections(sections: ResultSection[]): DocumentSection[] {
  return [...sections]
    .sort((a, b) => a.order - b.order)
    .map((section, index) => ({
      key: `${section.kind}-${section.order}`,
      number: String(index + 1).padStart(2, "0"),
      heading: section.heading,
      tag: cardLabel(section.kind),
      body: sectionText(section),
    }));
}
```

```python
def with_section_text(result: GenerationResult, order: int, text: str) -> GenerationResult | None:
    """[순수] 섹션 하나의 글을 바꾼 새 결과(복사본). 그 섹션이 없으면 None. 완성본(full_text)도 다시 만든다."""
    if not any(s.order == order for s in result.sections):

    sections = [
        s.model_copy(update={"blocks": [TextBlock(text=text)]}) if s.order == order else s.model_copy(deep=True)

    ]
    return result.model_copy(deep=True, update={"sections": sections, "full_text": compose_full_text(sections)})
```

```python
    def edit_section(self, state: SessionState, order: int, text: str) -> bool:
        """[코드] 결과 섹션 하나의 글을 바꾼 새 버전(edit)을 만든다. LLM을 부르지 않는다.

        직전 버전의 기준(사실·가정·미정)을 이어받고, 결과의 최신 여부(result_stale)는 그대로 둔다. 코드 검증만 다시
        돈다. 결과가 없거나 그 섹션이 없으면 False(세션은 그대로). 빈 글이면 HarnessError.
        """
        cleaned = text.strip()
        if not cleaned:
            raise HarnessError("고칠 문구가 비어 있습니다.")
        if state.result is None or not state.versions:
            return False
        result = with_section_text(state.result, order, cleaned)
        if result is None:
            return False
        report = ValidationReport(issues=collect_issues([*COMMON_VALIDATORS, *self.validators], state, result, self.policy))
        state.versions.append(state.versions[-1].derived(len(state.versions) + 1, VersionReason.EDIT, result, report))
        state.result = result
        state.validation = report
        return True
```

```python
def _values(items: Mapping[str, InfoItem]) -> dict[str, str]:
    return {name: item.value for name, item in items.items()}


def apply_supplements(current: Mapping[str, InfoItem], values: Mapping[str, str | None]) -> dict[str, InfoItem]:
    """[코드·순수] 보완 정보에 입력을 반영한 새 사전. 앞뒤 공백을 지우고, 빈 값·None이면 그 항목을 지운다."""
    merged = dict(current)
    for name, value in values.items():
        text = (value or "").strip()
        if not text:
            merged.pop(name, None)
        elif name not in merged or merged[name].value != text:
            merged[name] = InfoItem(value=text, source=InfoSource.USER_SUPPLEMENT)
    return merged


def supplements_changed(current: Mapping[str, InfoItem], latest: ResultVersion | None) -> bool:
    """[코드·순수] 보완 정보가 마지막 결과 버전에 반영된 것과 다른지(결과가 없으면 입력이 있는지)."""
    if latest is None:
        return bool(current)
    return _values(current) != _values(latest.supplements)


def generate_reason(state: SessionState) -> VersionReason:
    """[코드·순수] 이번 생성의 버전 이유. 답은 그대로이고 보완 정보만 바뀌었으면 보완 반영이다."""
    if not state.versions:
        return VersionReason.FIRST
    latest = state.versions[-1]
    same_answers = (
        _values(state.facts) == _values(latest.facts)
        and _values(state.assumptions) == _values(latest.assumptions)
        and list(state.unknowns) == latest.unknowns
    )
    if same_answers and supplements_changed(state.supplements, latest):
        return VersionReason.SUPPLEMENT
    return VersionReason.REGENERATE
```

```python
def new_version(state: SessionState, result: GenerationResult, report: ValidationReport) -> ResultVersion:
```

```ts
import type { SupplementItem } from "../api/types";

/** 미확인 정보 패널에 보일 것. canApply는 '초안에 반영'을 누를 수 있는지다. */
export type SupplementStatus = { line: string; missing: string; canApply: boolean };

/** 결과 문서에 실을 반영된 보완 정보 한 줄. */
export type AppliedRow = { label: string; value: string };

export const ALL_FILLED_NOTE = "모든 항목에 입력값이 있어요. 사실 검증 완료를 의미하지 않아요.";

const filled = (items: SupplementItem[]) => items.filter((item) => item.value !== null);

/**
 * [순수] 미확인 정보 패널 문구. current는 지금 입력(세션), applied는 지금 결과에 반영한 값(결과),
 * changed는 입력이 아직 결과에 반영되지 않았는지(서버가 정한 SessionView.supplements_changed)다.
 */
export function supplementStatus(current: SupplementItem[], applied: SupplementItem[], changed: boolean): SupplementStatus {
  const state = changed ? "아직 초안에 미반영" : filled(applied).length > 0 ? "초안에 반영됨" : "보완할 정보를 입력해 주세요";
  const missing = current.filter((item) => item.value === null).map((item) => item.label);
  return {
    line: `입력 ${filled(current).length}/${current.length} · ${state}`,
    missing: missing.length > 0 ? `미확인: ${missing.join(" · ")}` : ALL_FILLED_NOTE,
    canApply: changed,
  };
}

/** [순수] 대화상자 입력의 처음 값(항목 이름 → 글). 미확인은 빈 글이다. */
export function supplementDraft(items: SupplementItem[]): Record<string, string> {
  return Object.fromEntries(items.map((item) => [item.name, item.value ?? ""]));
}

/** [순수] 저장할 값. 앞뒤 공백을 지우고, 빈 글은 미확인(null)이다. */
export function supplementValues(draft: Record<string, string>): Record<string, string | null> {
  return Object.fromEntries(Object.entries(draft).map(([name, text]) => [name, text.trim() === "" ? null : text.trim()]));
}

/** [순수] 결과 문서에 실을 보완 정보. 반영한 값이 하나도 없으면 빈 목록이다(문서에 칸을 두지 않는다). */
export function appliedRows(applied: SupplementItem[]): AppliedRow[] {
  if (filled(applied).length === 0) return [];
  return applied.map((item) => ({ label: item.label, value: item.value ?? "미확인" }));
}
```

```tsx
import { useEffect, useId, useRef, useState } from "react";

import type { SupplementItem } from "../api/types";
import { supplementStatus } from "../lib/supplements";
import { SupplementForm } from "./SupplementForm";

type Props = {
  current: SupplementItem[]; // 지금 입력(세션)
  applied: SupplementItem[]; // 지금 결과에 반영한 값
  changed: boolean; // 입력이 아직 결과에 반영되지 않았다
  busy: boolean;
  onSave: (values: Record<string, string | null>) => void;
  onApply: () => void;
};

/**
 * 결과 오른쪽의 '미확인 정보 채우기'와 입력 대화상자. 대화상자가 열렸는지는 이 부품 안에서 끝나는 UI 상태라 여기서 가진다.
 * 닫으면(취소·저장·Esc) 연 버튼으로 포커스를 돌려준다. 보완 항목이 없는 과제면 그리지 않는다.
 */
export function SupplementPanel({ current, applied, changed, busy, onSave, onApply }: Props) {
  const [open, setOpen] = useState(false);
  const dialog = useRef<HTMLDialogElement>(null);
  const opener = useRef<HTMLButtonElement>(null);
  const titleId = useId();

  useEffect(() => {
    const node = dialog.current;
    if (node === null) return;
    if (open && !node.open) node.showModal();
    if (!open && node.open) node.close();
  }, [open]);

  if (current.length === 0) return null;
  const status = supplementStatus(current, applied, changed);
  const close = () => {
    setOpen(false);
    opener.current?.focus();
  };

  return (
    <section className="info-panel stack8" aria-label="미확인 정보 채우기">
      <h3>미확인 정보 채우기</h3>
      <p>{status.line}</p>
      <p className="small muted">{status.missing}</p>
      <div className="info-actions">
        <button ref={opener} type="button" className="btn" disabled={busy} onClick={() => setOpen(true)}>
          정보 입력·수정
        </button>
        <button type="button" className="btn" disabled={busy || !status.canApply} onClick={onApply}>
          초안에 반영
        </button>
      </div>
      <dialog
        ref={dialog}
        className="dialog"
        aria-labelledby={titleId}
        onCancel={(event) => {
          event.preventDefault(); // Esc도 같은 길(close)로 닫는다
          close();
        }}
      >
        {open && (
          <SupplementForm
            titleId={titleId}
            items={current}
            busy={busy}
            onSave={(values) => {
              onSave(values);
              close();
            }}
            onCancel={close}
          />
        )}
      </dialog>
    </section>
  );
}
```
