# 디자인 V2 묶음 2 (저장·버전·내 작업·결과물) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 세션을 서버 SQLite에 남기고, 결과가 바뀔 때마다 버전을 쌓아 저장·다시 보기·문서 내려받기를 하고, 내 작업·결과물 목록에서 이어서 작업하게 한다.

**Architecture:** 백엔드는 저장소 계약(`SessionStore`)에 SQLite 구현과 `list_by_owner`를 더하고, 생성(`generate`)이 버전(`ResultVersion`, 그때의 기준 포함)을 덧붙인다. 버전 보기·저장과 내 세션 요약은 얇은 라우터 + 순수 뷰 함수다. 프론트는 묶음 1의 해시 주소·리듀서·`needsLoad`·`screenModel` 구조에 화면(내 작업·결과물·저장 완료·버전 보기)과 데이터(세션 목록·버전)를 더하고, 문서 내려받기는 순수 함수가 만든 HTML을 저장한다.

**Tech Stack:** FastAPI · Pydantic v2 · sqlite3(표준 라이브러리) · pytest, React 19 · Vite 8 · TypeScript 5.9 · vitest 5 · openapi-fetch

## 설계 문서와 다르게(또는 더 구체적으로) 정한 것

1. **결과 카드는 그대로 둔다.** 섹션·블록(`TextBlock`·`ImageBlock`)으로 바꾸는 일은 이미지와 문구 수정이 들어오는 묶음 3에서 한다.
2. **버전은 그때의 기준을 함께 남긴다.** `ResultVersion`에 결과·검증과 함께 그 순간의 사실·가정·미정을 복사해 둔다. 결과 화면과 버전 보기의 '내 답변에서 반영한 기준'은 지금 세션이 아니라 그 버전의 기준이다(`ResultView.summary`·`unknowns`).
3. **버전 이유는 지금 `first`·`regenerate` 두 가지다.** `supplement`·`edit`·`evidence`는 그 기능이 생기는 묶음 3에서 더한다.
4. **결과물 목록은 `GET /sessions` 하나로 만든다.** 세션 요약에 저장한 버전(`saved_versions`)을 싣고, 프론트가 모은다.
5. **저장소 기본값은 `memory`다**(테스트가 결정적이도록). 로컬 개발은 `.env.example`이 `STORE=sqlite`로 켠다.
6. **'이 결과로 ○○ 시작'(다음 과제로 이어가기)은 묶음 4다.** 저장 완료 화면에는 아직 두지 않는다.

```python
"""세션 저장소. 메모리(테스트·임시)와 SQLite(서버를 다시 켜도 남는다) 두 가지다.

계약은 tests/test_store_contract.py가 두 구현 모두에 대해 확인한다.
"""
import sqlite3
from collections.abc import Iterator
from contextlib import closing, contextmanager
from pathlib import Path
from typing import Literal, Protocol

from app.core.results import utcnow
from app.core.state import SessionState

StoreKind = Literal["memory", "sqlite"]


class SessionStore(Protocol):
    """저장소 계약.

    - get은 복사본을 돌려준다. 받은 객체를 바꿔도 저장된 값은 그대로다. 없는 id면 None.
    - save는 복사본을 저장한다. 저장 뒤 원래 객체를 바꿔도 새지 않는다. 저장할 때 updated_at을 지금으로 바꾼다.
    - list_by_owner는 그 사용자의 세션 복사본을 최근에 저장한 순서로 돌려준다.
    """

    def get(self, session_id: str) -> SessionState | None: ...

    def save(self, state: SessionState) -> None: ...

    def list_by_owner(self, owner_id: str) -> list[SessionState]: ...


class InMemorySessionStore:
    """서버를 재시작하면 사라진다. 테스트의 기본값이다."""

    def __init__(self) -> None:
        self._sessions: dict[str, SessionState] = {}

    def get(self, session_id: str) -> SessionState | None:
        state = self._sessions.get(session_id)
        return state.model_copy(deep=True) if state else None

    def save(self, state: SessionState) -> None:
        state.updated_at = utcnow()
        self._sessions[state.session_id] = state.model_copy(deep=True)

    def list_by_owner(self, owner_id: str) -> list[SessionState]:
        owned = [s for s in self._sessions.values() if s.owner_id == owner_id]
        return [s.model_copy(deep=True) for s in sorted(owned, key=lambda s: s.updated_at, reverse=True)]


_SCHEMA = """
CREATE TABLE IF NOT EXISTS sessions (
    session_id TEXT PRIMARY KEY,
    owner_id TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    data TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS sessions_by_owner ON sessions (owner_id, updated_at);
"""


class SqliteSessionStore:
    """표 하나에 세션 JSON을 둔다(owner_id·updated_at은 목록을 위해 따로 둔다).

    요청마다 연결을 열고 닫으므로 스레드 풀에서 돌아도 안전하다. 같은 세션의 읽기→저장은 세션별 잠금
    (app/api/locks.py)이 하나로 묶으므로 서버 프로세스는 하나만 띄운다.
    """

    def __init__(self, path: Path) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        self._path = path
        with self._connect() as conn:
            conn.executescript(_SCHEMA)

    @contextmanager
    def _connect(self) -> Iterator[sqlite3.Connection]:
        """연결 하나에 트랜잭션 하나. 끝나면 커밋(예외면 롤백)하고 닫는다."""
        with closing(sqlite3.connect(self._path)) as conn:
            with conn:
                yield conn

    def get(self, session_id: str) -> SessionState | None:
        with self._connect() as conn:
            row = conn.execute("SELECT data FROM sessions WHERE session_id = ?", (session_id,)).fetchone()
        return SessionState.model_validate_json(row[0]) if row else None

    def save(self, state: SessionState) -> None:
        state.updated_at = utcnow()
        with self._connect() as conn:
            conn.execute(
                "INSERT INTO sessions (session_id, owner_id, updated_at, data) VALUES (?, ?, ?, ?) "
                "ON CONFLICT(session_id) DO UPDATE SET "
                "owner_id = excluded.owner_id, updated_at = excluded.updated_at, data = excluded.data",
                (state.session_id, state.owner_id, state.updated_at.isoformat(), state.model_dump_json()),
            )

    def list_by_owner(self, owner_id: str) -> list[SessionState]:
        with self._connect() as conn:
            rows = conn.execute(
                "SELECT data FROM sessions WHERE owner_id = ? ORDER BY updated_at DESC", (owner_id,)
            ).fetchall()
        return [SessionState.model_validate_json(row[0]) for row in rows]


def build_store(kind: StoreKind, path: Path) -> SessionStore:
    """설정(STORE·DATABASE_PATH)으로 저장소를 고른다."""
    return SqliteSessionStore(path) if kind == "sqlite" else InMemorySessionStore()
```

`updated_at`은 언제나 UTC(`+00:00`)라 ISO 문자열 순서가 시간 순서와 같다.

def to_result_view(state: SessionState, policy: TaskPolicy) -> ResultView:
    """[순수] 지금 결과. 기준(summary·unknowns)은 지금 세션이 아니라 이 결과를 만든 버전의 것이다."""
    if state.result is None or not state.versions:
        raise ValueError("생성된 결과가 없는 세션입니다.")
    latest = state.versions[-1]
    return ResultView(
        session_id=state.session_id,
        status=state.status,
        result=label_result(state.result, policy.labels),
        validation=state.validation,
        stale=state.result_stale,
        version=version_summary(latest),
        summary=build_summary(policy.fields, latest.facts, latest.assumptions),
        unknowns=label_unknowns(latest.unknowns, policy.labels),
    )


def to_version_view(state: SessionState, version: ResultVersion, policy: TaskPolicy) -> VersionView:

        session_id=state.session_id,
        version=version_summary(version),
        latest=version.version == state.versions[-1].version,
        result=label_result(version.result, policy.labels),
        validation=version.validation,

        unknowns=label_unknowns(version.unknowns, policy.labels),

/**
 * 화면 주소(해시). 주소를 읽고 만드는 곳은 parseRoute·toHash 두 함수뿐이다.
 * 과제 화면(요청·모드)은 세션 전이라 과제 이름으로, 세션 화면은 세션 id로 연다.
 */
export type Route =
  | { name: "home" }
  | { name: "guide" }
  | { name: "work" }
  | { name: "documents" }
  | { name: "request"; task: string }
  | { name: "mode"; task: string }
  | { name: "question"; sessionId: string }
  | { name: "revise"; sessionId: string; questionId: string }
  | { name: "confirm"; sessionId: string }
  | { name: "result"; sessionId: string }
  | { name: "saved"; sessionId: string }
  | { name: "version"; sessionId: string; version: number };

export const HOME: Route = { name: "home" };

const TASK_VIEWS = new Set(["request", "mode"]);
const VERSION_NUMBER = /^[1-9]\d*$/;

function decode(segment: string): string {
  try {
    return decodeURIComponent(segment);
  } catch {
    return segment;
  }
}

/** 조각 하나로 된 공용 화면. */
function page(name: string): Route | null {
  switch (name) {
    case "guide":
      return { name: "guide" };
    case "work":
      return { name: "work" };
    case "documents":
      return { name: "documents" };
    default:
      return null;
  }
}

/** `#/s/{id}/{view}` 모양의 세션 화면. */
function sessionView(sessionId: string, view: string): Route | null {
  switch (view) {
    case "question":
      return { name: "question", sessionId };
    case "confirm":
      return { name: "confirm", sessionId };
    case "result":
      return { name: "result", sessionId };
    case "saved":
      return { name: "saved", sessionId };
    default:
      return null;
  }
}

/** [순수] 해시를 주소로. 모르는 모양은 홈이다. */
export function parseRoute(hash: string): Route {
  const parts = hash.replace(/^#\/?/, "").split("/").filter(Boolean).map(decode);
  const [first, second, third, fourth] = parts;
  if (parts.length === 1) return page(first) ?? HOME;
  if (parts.length === 2 && first !== "s" && TASK_VIEWS.has(second)) {
    return second === "request" ? { name: "request", task: first } : { name: "mode", task: first };
  }
  if (first === "s" && parts.length === 3) return sessionView(second, third) ?? HOME;
  if (first === "s" && parts.length === 4 && third === "question") {
    return { name: "revise", sessionId: second, questionId: fourth };
  }
  if (first === "s" && parts.length === 4 && third === "v" && VERSION_NUMBER.test(fourth)) {
    return { name: "version", sessionId: second, version: Number(fourth) };
  }
  return HOME;
}

/** [순수] 화면 주소인지. 본문 바로가기 같은 문서 안 앵커(#main)는 화면 이동이 아니다. */
export function isRouteHash(hash: string): boolean {
  return hash === "" || hash === "#" || hash.startsWith("#/");
}

/** [순수] 주소를 해시로. */
export function toHash(route: Route): string {
  const e = encodeURIComponent;
  switch (route.name) {
    case "home":
      return "#/";
    case "guide":
    case "work":
    case "documents":
      return `#/${route.name}`;
    case "request":
    case "mode":
      return `#/${e(route.task)}/${route.name}`;
    case "revise":
      return `#/s/${e(route.sessionId)}/question/${e(route.questionId)}`;
    case "version":
      return `#/s/${e(route.sessionId)}/v/${route.version}`;
    default:
      return `#/s/${e(route.sessionId)}/${route.name}`;
  }
}

/** [순수] 화면 묶음. 같은 묶음 안에서 옮기면 기다리던 요청을 그대로 받고, 묶음을 떠나면 늦은 응답을 버린다. */
export function contextOf(route: Route): string {
  switch (route.name) {
    case "home":
    case "guide":
    case "work":
    case "documents":
      return route.name;
    case "request":
    case "mode":
      return `task:${route.task}`;
    default:
      return `session:${route.sessionId}`;
  }
}

export function isListRoute(route: Route): boolean {
  return route.name === "home" || route.name === "work" || route.name === "documents";
}

export function taskOf(route: Route): string | null {
  return route.name === "request" || route.name === "mode" ? route.task : null;
}

export function sessionIdOf(route: Route): string | null {
  return "sessionId" in route ? route.sessionId : null;

  return {
    route,
    tasks: null,
    sessions: null,
    form: null,
    session: null,
    sessionOutdated: false,
    result: null,
    version: null,
    drafts: {},
    inflight: null,
    stalled: null,
    toast: null,
  };
}

export function isBusy(state: AppState): boolean {
  return state.inflight !== null;
}

/** 앱 상태는 여기서만 바뀐다(순수). 응답은 지금 기다리는 요청(inflight.seq)의 것일 때만 받는다. */
export function reducer(state: AppState, action: Action): AppState {
  switch (action.type) {
    case "navigated":
      return navigate(state, action.route);
    case "requested":
      // 버튼 요청은 이전 알림을 지운다. 불러오기는 지우지 않는다(생성 실패 알림 뒤 곧바로 세션을 다시 받는다)
      return { ...state, inflight: { seq: action.seq, load: action.load }, toast: action.load === null ? null : state.toast };
    case "draftChanged":
      return { ...state, drafts: { ...state.drafts, [action.task]: action.text } };
    case "noticed":
      return { ...state, toast: action.message };
    case "toastClosed":
      return { ...state, toast: null };
    case "retried":
      return { ...state, stalled: null };
    default: {
      const inflight = state.inflight;
      if (inflight === null || inflight.seq !== action.seq) return state;
      return settle({ ...state, inflight: null }, inflight, action);
    }
  }
}

/**
 * 화면 이동. 과제·세션 묶음(contextOf)을 떠나면 기다리던 요청의 응답을 버리고, 다른 과제·세션의 데이터는 비운다.
 * 목록 화면에 새로 들어오면 내 세션 목록을 다시 받는다(그사이 답하거나 저장했을 수 있다).
 */
function navigate(state: AppState, route: Route): AppState {
  if (toHash(route) === toHash(state.route)) return state;
  const sessionId = sessionIdOf(route);
  const sameSession = state.session !== null && state.session.session_id === sessionId;
  return {
    ...state,
    route,
    inflight: contextOf(route) === contextOf(state.route) ? state.inflight : null,
    stalled: null,
    sessions: isListRoute(route) && !isListRoute(state.route) ? null : state.sessions,
    form: state.form !== null && state.form.task_type === taskOf(route) ? state.form : null,
    session: sameSession ? state.session : null,
    sessionOutdated: sameSession && state.sessionOutdated,
    result: state.result !== null && state.result.session_id === sessionId ? state.result : null,
    version: sameVersion(state.version, route) ? state.version : null,
  };
}

/** 이동한 주소가 지금 들고 있는 버전을 보는 주소인지. */
function sameVersion(version: VersionView | null, route: Route): boolean {
  return (
    version !== null &&
    route.name === "version" &&
    version.session_id === route.sessionId &&
    version.version.version === route.version
  );
}

function settle(done: AppState, inflight: Inflight, action: Settled): AppState {
  switch (action.type) {
    case "failed":
      return {
        ...done,
        toast: action.message,
        stalled: inflight.load ?? done.stalled,
        sessionOutdated: done.sessionOutdated || action.outdatesSession,
      };
    case "tasksLoaded":
      return { ...done, tasks: action.tasks };
    case "sessionsLoaded":
      return { ...done, sessions: action.sessions };
    case "formLoaded":
      return { ...done, form: action.form };
    case "sessionLoaded":
      return {
        ...done,
        route: action.next ?? done.route,
        session: action.session,
        sessionOutdated: false,
        result: matchingResult(done.result, action.session),
        toast: action.notice ?? done.toast,
      };
    case "resultLoaded":
      return {
        ...done,
        route: action.next ?? done.route,
        result: action.result,
        sessionOutdated: done.sessionOutdated || action.outdatesSession,
      };
    case "versionLoaded":
      return { ...done, version: action.version };
    case "versionSaved":
      // 지금 결과의 저장 표시가 바뀌었으므로 결과는 다시 받는다
      return { ...done, route: action.next ?? done.route, version: action.version, result: null };
  }
}

/** 받은 세션에 맞는 결과만 남긴다(다른 세션이거나 결과가 없어졌으면 버린다). */
function matchingResult(result: ResultView | null, session: SessionView): ResultView | null {
  return result !== null && result.session_id === session.session_id && session.has_result ? result : null;
}

import { isListRoute, sessionIdOf, taskOf } from "../lib/routes";
import type { AppState } from "./state";

/** 화면을 그리는 데 모자란 데이터 하나. */
export type Load =
  | { kind: "tasks" }
  | { kind: "sessions" }
  | { kind: "form"; task: string }
  | { kind: "session"; sessionId: string }
  | { kind: "result"; sessionId: string }
  | { kind: "version"; sessionId: string; version: number };

export function loadKey(load: Load): string {
  switch (load.kind) {
    case "tasks":
    case "sessions":
      return load.kind;
    case "form":
      return `form:${load.task}`;
    case "version":
      return `version:${load.sessionId}:${load.version}`;
    default:
      return `${load.kind}:${load.sessionId}`;
  }
}

/** [순수] 지금 불러올 데이터. 요청을 기다리는 중이거나 같은 불러오기가 이 화면에서 실패했으면 부르지 않는다. */
export function needsLoad(state: AppState): Load | null {
  if (state.inflight !== null) return null;
  const load = missing(state);
  return load !== null && loadKey(load) !== state.stalled ? load : null;
}

function missing(state: AppState): Load | null {
  const { route } = state;
  if (state.tasks === null) return { kind: "tasks" };
  if (isListRoute(route)) return state.sessions === null ? { kind: "sessions" } : null;
  const task = taskOf(route);
  if (task !== null) return state.form?.task_type === task ? null : { kind: "form", task };
  const sessionId = sessionIdOf(route);
  if (sessionId === null) return null;
  const session = state.session;
  if (session === null || session.session_id !== sessionId || state.sessionOutdated) {
    return { kind: "session", sessionId };
  }
  if (route.name === "version") {
    const version = state.version;
    const loaded = version !== null && version.session_id === sessionId && version.version.version === route.version;
    return loaded ? null : { kind: "version", sessionId, version: route.version };
  }
  const wantsResult = (route.name === "result" || route.name === "saved") && session.has_result;
  return wantsResult && state.result?.session_id !== sessionId ? { kind: "result", sessionId } : null;
}

} from "../api/types";
import { isEditable } from "../lib/flowView";
import type { EditableFlowItem } from "../lib/flowView";
import type { AppState } from "./state";

/** 지금 그릴 화면과 그 화면에 필요한 값. 화면은 이 값만 보고 그린다. */
export type ScreenModel =
  | { kind: "loading" }
  | { kind: "unavailable" }
  | { kind: "home"; tasks: TaskSummary[]; sessions: SessionSummary[] }
  | { kind: "guide" }
  | { kind: "work"; sessions: SessionSummary[] }
  | { kind: "documents"; sessions: SessionSummary[] }
  | { kind: "request"; task: TaskSummary; form: TaskForm; draft: string }
  | { kind: "mode"; task: TaskSummary; form: TaskForm; draft: string }
  | { kind: "question"; task: TaskSummary; session: SessionView; question: Question }
  | { kind: "revise"; task: TaskSummary; session: SessionView; item: EditableFlowItem }
  | { kind: "confirm"; task: TaskSummary; session: SessionView }
  | { kind: "result"; task: TaskSummary; session: SessionView; result: ResultView | null }
  | { kind: "saved"; task: TaskSummary; session: SessionView; result: ResultView }
  | { kind: "version"; task: TaskSummary; session: SessionView; version: VersionView };

const LOADING: ScreenModel = { kind: "loading" };

/** [순수] 지금 그릴 화면. 데이터가 모자라면 loading, 불러오기가 실패했으면 unavailable이다. */
export function screenModel(state: AppState): ScreenModel {
  const model = resolve(state);
  return model.kind === "loading" && state.stalled !== null ? { kind: "unavailable" } : model;
}

function resolve(state: AppState): ScreenModel {
  const { route } = state;
  switch (route.name) {
    case "home":
      return state.tasks === null || state.sessions === null
        ? LOADING
        : { kind: "home", tasks: state.tasks, sessions: state.sessions };
    case "guide":
      return { kind: "guide" };
    case "work":
    case "documents":
      if (state.sessions === null) return LOADING;
      return route.name === "work" ? { kind: "work", sessions: state.sessions } : { kind: "documents", sessions: state.sessions };
    case "request":
    case "mode": {
      const task = findTask(state.tasks, route.task);
      if (task === null || state.form === null || state.form.task_type !== route.task) return LOADING;
      const data = { task, form: state.form, draft: state.drafts[route.task] ?? "" };
      return route.name === "request" ? { kind: "request", ...data } : { kind: "mode", ...data };
    }
    default: {
      const session = state.session;
      if (session === null || session.session_id !== route.sessionId) return LOADING;
      const task = findTask(state.tasks, session.task_type);
      if (task === null) return LOADING;
      if (route.name === "version") {
        const version = state.version;
        const loaded = version !== null && version.session_id === session.session_id && version.version.version === route.version;
        return loaded ? { kind: "version", task, session, version } : LOADING;
      }
      if (route.name === "result" || route.name === "saved") {
        if (!session.has_result) return { kind: "result", task, session, result: null };
        const result = state.result !== null && state.result.session_id === session.session_id ? state.result : null;
        if (result === null) return LOADING;
        return route.name === "saved" ? { kind: "saved", task, session, result } : { kind: "result", task, session, result };
      }
      if (route.name === "revise") {
        const item = session.flow.find((i) => i.question?.question_id === route.questionId);
        if (item !== undefined && isEditable(item)) return { kind: "revise", task, session, item };
      }
      if (route.name !== "confirm" && session.next_question !== null) {
        return { kind: "question", task, session, question: session.next_question };
      }
      return { kind: "confirm", task, session };
    }
  }
}

function findTask(tasks: TaskSummary[] | null, taskType: string): TaskSummary | null {
  return tasks?.find((t) => t.task_type === taskType) ?? null;
}

import type { LabeledResult, ResultCard, ResultView, TaskSummary, ValidationIssue, VersionSummary } from "../api/types";
import { criteriaRows } from "./criteria";
import type { CriteriaRow } from "./criteria";
import { cardLabel } from "./labels";
import { versionLine } from "./sessions";

export type DocumentSection = { key: string; number: string; heading: string; tag: string; body: string };

/** [순수] 결과 카드를 문서 섹션(순서대로 01, 02 …)으로. */
export function documentSections(cards: ResultCard[]): DocumentSection[] {
  return [...cards]
    .sort((a, b) => a.order - b.order)
    .map((card, index) => ({
      key: `${card.card_type}-${card.order}`,
      number: String(index + 1).padStart(2, "0"),
      heading: card.title,
      tag: cardLabel(card.card_type),
      body: card.body,
    }));
}

export function resultTitle(taskTitle: string): string {
  return `${taskTitle} 초안`;
}

/** 결과 문서 위 작은 줄: "V2 · 다시 만든 초안 / 결과물 이름". 지금 답과 맞지 않으면 함께 알린다. */
export function resultKicker(deliverable: string, version: Pick<VersionSummary, "version" | "reason">, stale: boolean): string {
  const line = `${versionLine(version)} / ${deliverable}`;
  return stale ? `${line} · 최신 아님` : line;
}

/** 문서로 보여 줄 결과. 지금 결과(ResultView)와 지난 버전(VersionView)이 같은 모양이다. */
export type Shown = Pick<ResultView, "result" | "validation" | "version" | "summary" | "unknowns">;

export type DocumentContent = {
  title: string;
  kicker: string;
  rows: CriteriaRow[];
  result: LabeledResult;
  issues: ValidationIssue[];
};

/** [순수] 화면의 결과 문서와 내려받는 문서가 같이 쓰는 내용. 기준은 그 결과를 만든 버전의 것이다. */
export function documentContent(
  task: Pick<TaskSummary, "title" | "deliverable">,
  shown: Shown,
  stale: boolean,
): DocumentContent {
  return {
    title: resultTitle(task.title),
    kicker: resultKicker(task.deliverable, shown.version, stale),
    rows: criteriaRows(shown.summary, shown.unknowns),
    result: shown.result,
    issues: shown.validation?.issues ?? [],
  };
}

import { formatAssumption } from "./assumption";
import type { DocumentContent } from "./document";
import { documentSections } from "./document";

export type DocumentFile = { filename: string; html: string };

const ESCAPES: Record<string, string> = { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" };

export function escapeHtml(text: string): string {
  return text.replace(/[&<>"']/g, (char) => ESCAPES[char]);
}

// 내려받은 파일은 앱 밖에서 열리므로 스타일을 파일 안에 둔다(인쇄해도 읽기 좋게).
const STYLE = [
  'body{font-family:"Noto Sans KR",sans-serif;max-width:760px;margin:40px auto;padding:0 16px;line-height:1.7;color:#202020}',
  "h1{font-size:28px;margin:4px 0 24px}h2{font-size:20px;margin:28px 0 8px}",
  "p{white-space:pre-line;margin:0 0 8px}.note{color:#707070;font-size:13px}.num{color:#800122;font-size:13px}",
  "table{border-collapse:collapse;width:100%}th,td{padding:6px 8px;border-top:1px solid #e0e0e0;text-align:left;vertical-align:top}",
  "th{width:120px;font-weight:500}",
].join("");

function list(heading: string, items: string[]): string {
  if (items.length === 0) return "";
  return `<h2>${escapeHtml(heading)}</h2><ul>${items.map((item) => `<li>${escapeHtml(item)}</li>`).join("")}</ul>`;
}

/** [순수] 내려받을 결과 문서(HTML 한 파일). 사용자·AI가 쓴 글은 모두 이스케이프한다. 이미지가 들어갈 수 있어 HTML이다. */
export function documentHtml(content: DocumentContent & { request: string }): string {
  const e = escapeHtml;
  const rows = content.rows
    .map((row) => `<tr><th>${e(row.label)}</th><td>${e(row.value)}${row.note ? ` <span class="note">· ${e(row.note)}</span>` : ""}</td></tr>`)
    .join("");
  const sections = documentSections(content.result.cards)
    .map((s) => `<h2><span class="num">${s.number}</span> ${e(s.heading)} <span class="note">${e(s.tag)}</span></h2><p>${e(s.body)}</p>`)
    .join("");
  return [
    "<!doctype html>",
    '<html lang="ko">',
    "<head>",
    '<meta charset="utf-8">',
    `<title>${e(content.title)}</title>`,
    `<style>${STYLE}</style>`,
    "</head>",
    "<body>",
    `<p class="note">${e(content.kicker)}</p>`,
    `<h1>${e(content.title)}</h1>`,
    "<h2>내 요청</h2>",
    `<p>${e(content.request)}</p>`,
    "<h2>내 답변에서 반영한 기준</h2>",
    `<table>${rows}</table>`,
    sections,
    list("AI가 가정한 내용", content.result.assumptions.map(formatAssumption)),
    list("검토가 필요한 부분", content.issues.map((issue) => issue.message)),
    '<p class="note">창작자 AX에서 만든 초안이에요. AI 제안과 외부 사실은 따로 검토가 필요해요.</p>',
    "</body>",
    "</html>",
  ].join("\n");
}

import type { TaskSummary } from "../api/types";
import { documentContent } from "../lib/document";
import type { Shown } from "../lib/document";
import { documentHtml } from "../lib/documentHtml";
import type { DocumentFile } from "../lib/documentHtml";
import type { ScreenModel } from "./screenModel";

function fileFor(task: TaskSummary, request: string, shown: Shown, stale: boolean): DocumentFile {
  return {
    filename: `AX_${task.task_type}_V${shown.version.version}.html`,
    html: documentHtml({ ...documentContent(task, shown, stale), request }),
  };
}

/** [순수] 지금 화면에서 내려받을 문서. 결과가 보이는 화면(결과·저장 완료·버전 보기)에만 있다. */
export function documentFileOf(model: ScreenModel): DocumentFile | null {
  switch (model.kind) {
    case "result":
      return model.result === null ? null : fileFor(model.task, model.session.raw_input, model.result, model.session.result_stale);
    case "saved":
      return fileFor(model.task, model.session.raw_input, model.result, model.session.result_stale);
    case "version":
      return fileFor(model.task, model.session.raw_input, model.version, !model.version.latest);
    default:
      return null;
  }
}

export function ResultDocument({ title, kicker, rows, result, issues }: DocumentContent) {
  const { assumptions } = result;
  return (
    <article className="card document stack">
      <div className="doc-title stack8">
        <p className="small muted">{kicker}</p>
        <h1>{title}</h1>

          <h2 className="section-heading">AI가 가정한 내용</h2>

export type Handlers = {

  guide: () => void;
  goHome: () => void;
  retry: () => void;
  start: (task: string) => void;
  changeDraft: (task: string, text: string) => void;
  submitRequest: (task: string) => void;
  editRequest: (task: string) => void;
  pickMode: (task: string, mode: CoachingMode) => void;
  answer: (body: AnswerBody) => void;
  example: (action: ExampleAction["action"]) => void;
  earlyDraft: () => void;
  edit: (questionId: string) => void;
  revise: (questionId: string, body: ReviseBody) => void;

  generate: () => void;

/** 화면 모델 → 본문과 오른쪽 패널. 판단은 하지 않고 모델과 lib의 순수 함수로만 그린다. */
export function renderScreen(model: ScreenModel, busy: boolean, on: Handlers): ScreenParts {
  const file = documentFileOf(model);
  const download = () => {
    if (file) on.download(file);
  };
  switch (model.kind) {
    case "loading":
      return { main: <LoadingScreen />, side: null };
    case "unavailable":
      return { main: <UnavailableScreen onRetry={on.retry} onHome={on.goHome} />, side: null };
    case "home":
      return {
        main: (
          <HomeScreen
            tasks={model.tasks}
            sessions={model.sessions}
            onStart={on.start}
            onResume={on.resume}
            onWork={() => on.navigate("work")}
          />
        ),
        side: null,
      };
    case "guide":
      return { main: <GuideScreen onStart={on.goHome} />, side: null };
    case "work":
      return { main: <WorkScreen sessions={model.sessions} onResume={on.resume} onHome={on.goHome} />, side: null };
    case "documents":
      return { main: <DocumentsScreen sessions={model.sessions} onOpen={on.openVersion} onHome={on.goHome} />, side: null };
    case "request": {
      const task = model.task.task_type;
      return {
        main: (
          <RequestScreen
            task={model.task}
            form={model.form}
            draft={model.draft}
            onChange={(text) => on.changeDraft(task, text)}
            onSubmit={() => on.submitRequest(task)}
          />
        ),
        side: null,
      };
    }
    case "mode": {
      const task = model.task.task_type;
      return {
        main: (
          <ModeScreen
            form={model.form}
            draft={model.draft}
            busy={busy}
            onPick={(mode) => on.pickMode(task, mode)}
            onEditRequest={() => on.editRequest(task)}
          />
        ),
        side: null,
      };
    }
    case "question":
      return {
        main: (
          <QuestionCard
            key={model.question.question_id}
            question={model.question}
            kicker={questionKicker(model.question, answeredBefore(model.session.flow))}
            request={model.session.raw_input}
            busy={busy}
            onAnswer={on.answer}
            onExample={on.example}
            onEarlyDraft={on.earlyDraft}
          />
        ),
        side: flowSide(model.session, false, busy, on),
      };
    case "revise": {
      const questionId = model.item.question.question_id;
      return {
        main: (

            key={questionId}
            item={model.item}
            busy={busy}

            onCancel={on.backToQuestions}
          />
        ),
        side: flowSide(model.session, false, busy, on),
      };
    }
    case "confirm": {
      const editable = firstEditable(model.session.flow);
      return {
        main: (
          <ConfirmScreen
            task={model.task}
            session={model.session}
            busy={busy}
            onAct={(action) => (action === "generate" ? on.generate() : on.showResult())}
            onChangeAnswers={editable ? () => on.edit(editable) : null}
            onBackToQuestion={model.session.next_question ? on.backToQuestions : null}
          />
        ),
        side: flowSide(model.session, true, busy, on),
      };
    }
    case "result": {
      const editable = firstEditable(model.session.flow);
      return {
        main: <ResultMain task={model.task} session={model.session} result={model.result} busy={busy} onConfirm={on.confirm} />,
        side:

              session={model.session}
              version={model.result.version}
              busy={busy}
              onRegenerate={on.generate}
              onSave={on.saveVersion}
              onChangeAnswers={editable ? () => on.edit(editable) : null}
              onContinue={on.continueQuestions}
              onDownload={download}
            />
          ),
      };
    }
    case "saved":
      return {
        main: (
          <SavedMain
            task={model.task}

            onResult={on.showResult}

      flow={session.flow}

      side={side}

    </AppShell>
  );
}

import type { DocumentFile } from "./documentHtml";

/** 브라우저에서 파일로 내려받는다. 화면 밖 부수 효과라 App이 handlers에 넘긴다(테스트는 가짜를 넘긴다). */
export function saveFile({ filename, html }: DocumentFile): void {
  const url = URL.createObjectURL(new Blob([html], { type: "text/html;charset=utf-8" }));
  const link = document.createElement("a");
  link.href = url;
  link.download = filename;
  link.click();
  window.setTimeout(() => URL.revokeObjectURL(url), 1000);
}
